// ── TÜRKÇE YARDIMCI ──

const SESLI = /[aeıiouöüAEIİOUÖÜ]/g;


function heceHesapla(w) {

  const m = w.match(SESLI);

  return m ? m.length : 1;

}


function sozcukBol(t) {

  return t.match(/[a-zA-ZğüşıöçĞÜŞİÖÇ']+/g) || [];

}


function cumleBol(t) {

  // Noktadan önce kısaltmalar gelirse de bölme

  return t.split(/(?<=[.!?…])\s+/).map(s => s.trim()).filter(s => sozcukBol(s).length > 0);

}


function paragrafSay(t) {

  return t.split(/\n+/).map(s => s.trim()).filter(s => s.length > 0).length;

}


// ── HESAPLAMA ──

function hesapla() {

  const txt = document.getElementById('txt').value;


  if (!txt.trim()) {

    ['s-kar','s-soz','s-cum','s-par','s-x1','s-x2'].forEach(id => {

      document.getElementById(id).textContent = '—';

    });

    ['sc-atesman','sc-cetinkaya','sc-bezirci','sc-sonmez'].forEach(id => {

      document.getElementById(id).textContent = '—';

    });

    ['lv-atesman','lv-cetinkaya','lv-bezirci','lv-sonmez'].forEach(id => {

      document.getElementById(id).textContent = '—';

    });

    document.getElementById('freq-area').innerHTML = '<div class="empty">Metin girin, en sık sözcükler burada görünecek.</div>';

    return;

  }


  const cumler  = cumleBol(txt);

  const sozler  = sozcukBol(txt);

  const karSay  = txt.length;

  const sozSay  = sozler.length;

  const cumSay  = Math.max(cumler.length, 1);

  const parSay  = paragrafSay(txt);


  document.getElementById('s-kar').textContent = karSay.toLocaleString('tr-TR');

  document.getElementById('s-soz').textContent = sozSay.toLocaleString('tr-TR');

  document.getElementById('s-cum').textContent = cumSay.toLocaleString('tr-TR');

  document.getElementById('s-par').textContent = parSay.toLocaleString('tr-TR');


  if (sozSay === 0) return;


  // Hece sayıları

  let toplamHece = 0, h3 = 0, h4 = 0, h5 = 0, h6 = 0;

  sozler.forEach(w => {

    const h = heceHesapla(w);

    toplamHece += h;

    if (h === 3) h3++;

    else if (h === 4) h4++;

    else if (h === 5) h5++;

    else if (h >= 6) h6++;

  });


  const x1 = toplamHece / sozSay;         // ort. hece/sözcük

  const x2 = sozSay / cumSay;             // ort. sözcük/cümle

  const OKS = x2;                          // cümledeki ort. kelime = x2


  document.getElementById('s-x1').textContent = x1.toFixed(2);

  document.getElementById('s-x2').textContent = x2.toFixed(2);


  // 1. ATEŞMAN

  const pAtesman = 198.825 - 40.175 * x1 - 2.610 * x2;

  document.getElementById('sc-atesman').textContent = Math.round(pAtesman);

  document.getElementById('lv-atesman').textContent = lvAtesman(pAtesman);


  // 2. ÇETİNKAYA-UZUN

  const pCetinkaya = 118.823 - 25.987 * x1 - 0.971 * x2;

  document.getElementById('sc-cetinkaya').textContent = Math.round(pCetinkaya);

  document.getElementById('lv-cetinkaya').textContent = lvCetinkaya(pCetinkaya);


  // 3. BEZİRCİ-YILMAZ

  // H3, H4, H5, H6 are averages per sentence

  const H3 = h3 / cumSay;

  const H4 = h4 / cumSay;

  const H5 = h5 / cumSay;

  const H6 = h6 / cumSay;

  const inner = OKS * (H3 * 0.84 + H4 * 1.5 + H5 * 3.5 + H6 * 26.25);

  const pBezirci = inner > 0 ? Math.sqrt(inner) : 0;

  document.getElementById('sc-bezirci').textContent = pBezirci.toFixed(2);

  document.getElementById('lv-bezirci').textContent = lvBezirci(pBezirci);


  // 4. SÖNMEZ

  const yabanciSay = parseInt(document.getElementById('yabanciSay').value) || 0;

  const pSonmez = ((sozSay - yabanciSay) / sozSay) * 100;

  document.getElementById('sc-sonmez').textContent = Math.round(pSonmez) + '%';

  document.getElementById('lv-sonmez').textContent = lvSonmez(pSonmez);


  // Frekans

  renderFreq(sozler);

}


// ── SEVIYE FONKSİYONLARI ──

function lvAtesman(p) {

  if (p >= 90) return 'Çok Kolay';

  if (p >= 70) return 'Kolay';

  if (p >= 50) return 'Orta Güçlükte';

  if (p >= 30) return 'Güç';

  return 'Çok Güç';

}

function lvCetinkaya(p) {

  if (p >= 65) return 'Bağımsız Okuma';

  if (p >= 50) return 'Kolay';

  if (p >= 35) return 'Eğitsel Okuma';

  return 'Engelli Okuma';

}

function lvBezirci(p) {

  if (p <= 0) return '—';

  if (p <= 8) return 'İlköğretim';

  if (p <= 12) return 'Ortaöğretim';

  if (p <= 16) return 'Lisans';

  return 'Akademik';

}

function lvSonmez(p) {

  if (p >= 75) return 'Bağımsız Okuma';

  if (p >= 50) return 'Eğitsel Okuma';

  return 'Engelli Okuma';

}


// ── FREKANS ──

function renderFreq(sozler) {

  const freq = {};

  sozler.forEach(w => { const k = w.toLowerCase(); freq[k] = (freq[k] || 0) + 1; });

  const top = Object.entries(freq).sort((a,b) => b[1] - a[1]).slice(0, 15);

  if (top.length === 0) return;

  const max = top[0][1];

  let rows = top.map(([w, c]) =>

    `<tr>

      <td>${w}</td>

      <td style="text-align:right;color:var(--ink-2)">${c}</td>

      <td><div class="freq-bar-bg"><div class="freq-bar" style="width:${Math.round(c/max*100)}%"></div></div></td>

    </tr>`).join('');

  document.getElementById('freq-area').innerHTML =

    `<table class="freq-table">

      <thead><tr><th>Sözcük</th><th>Sıklık</th><th>Dağılım</th></tr></thead>

      <tbody>${rows}</tbody>

    </table>`;

}
